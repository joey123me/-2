using System;

// Базовый класс
public class Transport
{
    protected string TransportId;
    protected int CurrentSpeed;
    private double _mileageKm;

    public string Id => TransportId;
    public double Mileage => _mileageKm;
    public int Speed
    {
        get => CurrentSpeed;
        protected set => CurrentSpeed = Math.Max(0, value);
    }

    public Transport(string id, double mileage)
    {
        TransportId = id;
        _mileageKm = mileage;
        CurrentSpeed = 0;
    }

    public virtual void Move(double distanceKm)
    {
        _mileageKm += distanceKm;
        Console.WriteLine($"[{Id}] Пройдено {distanceKm:F1} км. Общий пробег: {_mileageKm:F1} км.");
    }

    public virtual void DisplayInfo()
    {
        Console.WriteLine($"Транспорт: {Id}, скорость: {Speed} км/ч, пробег: {Mileage:F1} км");
    }
}

// Городской автобус
public class PublicBus : Transport
{
    public int PassengerCapacity { get; }
    public int CurrentPassengers { get; private set; }

    public PublicBus(string id, double mileage, int passengerCapacity)
        : base(id, mileage)
    {
        PassengerCapacity = passengerCapacity;
        CurrentPassengers = 0;
    }

    public void BoardPassengers(int count)
    {
        if (count < 0)
        {
            Console.WriteLine("Нельзя посадить отрицательное количество пассажиров.");
            return;
        }

        int spaceLeft = PassengerCapacity - CurrentPassengers;
        int toBoard = Math.Min(count, spaceLeft);

        CurrentPassengers += toBoard;
        if (toBoard < count)
        {
            Console.WriteLine($"Автобус заполнен. Посажено только {toBoard} из {count} пассажиров.");
        }
        else
        {
            Console.WriteLine($"Посажено {toBoard} пассажиров. Всего в салоне: {CurrentPassengers}/{PassengerCapacity}");
        }
    }

    public override void Move(double distanceKm)
    {
        base.Move(distanceKm);
        Console.WriteLine("[PublicBus] Автобус следует по городскому маршруту с остановками.");
    }
}

// Электросамокат (запечатанный класс)
public sealed class ElectricScooter : Transport
{
    private int _batteryPercent;

    public ElectricScooter(string id, double mileage, int batteryPercent)
        : base(id, mileage)
    {
        _batteryPercent = Math.Clamp(batteryPercent, 0, 100);
    }

    public void Recharge()
    {
        _batteryPercent = 100;
        Console.WriteLine($"[{Id}] Заряд восстановлен до 100%.");
    }

    public override void Move(double distanceKm)
    {
        int requiredCharge = (int)Math.Ceiling(distanceKm * 2); // 2% на 1 км

        if (_batteryPercent < requiredCharge)
        {
            Console.WriteLine($"[{Id}] Недостаточно заряда для проезда {distanceKm:F1} км (нужно {requiredCharge}%, есть {_batteryPercent}%). Движение заблокировано.");
            return;
        }

        _batteryPercent -= requiredCharge;
        base.Move(distanceKm);
        Console.WriteLine($"[ElectricScooter] Заряд после поездки: {_batteryPercent}%");
    }
}

// Грузовой тягач
public class CargoTruck : Transport
{
    public double MaxCargoTons { get; }
    public bool HasTrailer { get; set; }

    public CargoTruck(string id, double mileage, double maxCargoTons, bool hasTrailer)
        : base(id, mileage)
    {
        MaxCargoTons = maxCargoTons;
        HasTrailer = hasTrailer;
    }

    public override void DisplayInfo()
    {
        base.DisplayInfo();
        string trailerStatus = HasTrailer ? "с прицепом" : "без прицепа";
        Console.WriteLine($"[CargoTruck] Грузоподъёмность: {MaxCargoTons} т, статус: {trailerStatus}");
    }

    public void UnloadCargo()
    {
        Console.WriteLine($"[{Id}] Производится выгрузка груза. Грузоподъёмность: {MaxCargoTons} т.");
    }
}

class Program
{
    static void Main()
    {
        // Полиморфный массив
        Transport[] fleet = new Transport[4]
        {
            new PublicBus("BUS-001", 12345.6, 50),
            new ElectricScooter("SCOOT-001", 89.2, 30),
            new ElectricScooter("SCOOT-002", 45.7, 80),
            new CargoTruck("TRUCK-001", 98765.4, 20.0, true)
        };

        // Проезд 10 км для всех
        foreach (var item in fleet)
        {
            item.Move(10.0);
            Console.WriteLine();
        }

        // Демонстрация is + проверка заряда и подзарядка
        foreach (var item in fleet)
        {
            if (item is ElectricScooter scooter)
            {
                Console.WriteLine($"[Проверка] Самокат {scooter.Id}, заряд: {GetBatteryPercent(scooter)}%");
                if (GetBatteryPercent(scooter) < 20)
                {
                    scooter.Recharge();
                }
                Console.WriteLine();
            }
        }

        // Демонстрация as + выгрузка для грузовиков
        foreach (var item in fleet)
        {
            CargoTruck? truck = item as CargoTruck;
            if (truck != null)
            {
                truck.UnloadCargo();
            }
        }

        // Попытка наследования от sealed-класса (закомментировано, чтобы код компилировался)
        // public class IllegalInheritance : ElectricScooter { } 
        // Компилятор выдаст ошибку: "Cannot derive from sealed class 'ElectricScooter'"
    }

    // Вспомогательный метод, чтобы получить заряд (т.к. поле private)
    private static int GetBatteryPercent(ElectricScooter scooter)
    {
        var bindingFlags = System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Instance;
        var field = typeof(ElectricScooter).GetField("_batteryPercent", bindingFlags);
        return (int)field?.GetValue(scooter) ?? -1;
    }
}
